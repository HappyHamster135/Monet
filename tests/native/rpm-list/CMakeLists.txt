cmake_minimum_required(VERSION 3.22)
project(MonetRpmListTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6FileMetaData REQUIRED)
add_executable(rpm-list main.cpp)
target_include_directories(rpm-list PRIVATE ${ARK_SOURCE}/kerfuffle ${ARK_BUILD}/kerfuffle)
target_link_libraries(rpm-list PRIVATE Qt6::Core Qt6::Gui KF6::CoreAddons KF6::KIOCore KF6::FileMetaData /usr/lib64/libkerfuffle.so.26)
