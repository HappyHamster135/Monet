cmake_minimum_required(VERSION 3.16)
project(MonetSettingsStyle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(monetsettings MODULE monetsettings.cpp)
target_link_libraries(monetsettings PRIVATE Qt6::Widgets)
set_target_properties(monetsettings PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/styles")
add_executable(settings-layout-test layout-test.cpp)
target_link_libraries(settings-layout-test PRIVATE Qt6::Widgets Qt6::Test)
enable_testing()
add_test(NAME settings-layout COMMAND settings-layout-test)
set_tests_properties(settings-layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_BINARY_DIR}/plugins")
