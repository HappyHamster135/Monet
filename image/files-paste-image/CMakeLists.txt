cmake_minimum_required(VERSION 3.22)
project(MonetFilesPasteImage LANGUAGES CXX)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Concurrent Widgets)
set(QT_MAJOR_VERSION 6)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
set(KDE_INSTALL_USE_QT_SYS_PATHS ON)
include(KDEInstallDirs6)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
find_package(KF6CoreAddons REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(KF6KIO REQUIRED)
find_package(KF6WidgetsAddons REQUIRED)
kcoreaddons_add_plugin(monet_paste_image SOURCES pasteimage.cpp INSTALL_NAMESPACE "kf6/kfileitemaction")
target_link_libraries(monet_paste_image PRIVATE Qt6::Concurrent Qt6::Widgets KF6::CoreAddons KF6::I18n KF6::KIOWidgets KF6::WidgetsAddons)
install(FILES pasteimage.cpp pasteimage.json CMakeLists.txt DESTINATION ${KDE_INSTALL_DATADIR}/doc/monet-files-paste-image)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(pasteimagetest tests/pasteimagetest.cpp)
    target_link_libraries(pasteimagetest PRIVATE Qt6::Test Qt6::Widgets KF6::KIOWidgets)
    add_test(NAME files-paste-image COMMAND pasteimagetest)
    set_tests_properties(files-paste-image PROPERTIES TIMEOUT 25 ENVIRONMENT
        "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${CMAKE_LIBRARY_OUTPUT_DIRECTORY};LANGUAGE=en;LC_ALL=C.UTF-8")
endif()
install(FILES tests/pasteimagetest.cpp DESTINATION ${KDE_INSTALL_DATADIR}/doc/monet-files-paste-image/tests)
