cmake_minimum_required(VERSION 3.22)
project(MonetActivity LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Qml)
find_package(KWayland CONFIG REQUIRED)
add_library(monetactivityplugin MODULE plugin.cpp activity.cpp activity.h monitor.cpp monitor.h power.cpp power.h)
target_link_libraries(monetactivityplugin PRIVATE Qt6::Core Qt6::DBus Qt6::Qml Plasma::KWaylandClient)
target_compile_options(monetactivityplugin PRIVATE -Wall -Wextra -Werror)
set_target_properties(monetactivityplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/org/monet/activity")
configure_file(qmldir "${CMAKE_CURRENT_BINARY_DIR}/qml/org/monet/activity/qmldir" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
