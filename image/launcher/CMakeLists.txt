cmake_minimum_required(VERSION 3.22)
project(MonetLauncher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Qml Quick)
find_package(KF6KIO REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6Notifications REQUIRED)
find_package(KF6I18n REQUIRED)
find_package(LibTaskManager REQUIRED)
find_package(PlasmaActivities REQUIRED)
add_library(monetlaunchcore STATIC coordinator.cpp coordinator.h folderopener.cpp folderopener.h)
set_target_properties(monetlaunchcore PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_link_libraries(monetlaunchcore PUBLIC Qt6::Core Qt6::DBus Qt6::Quick KF6::KIOGui KF6::KIOCore KF6::Service
    KF6::WindowSystem KF6::Notifications KF6::I18n PW::LibTaskManager Plasma::Activities)
target_compile_options(monetlaunchcore PRIVATE -Wall -Wextra -Werror)
add_library(monetlauncherplugin MODULE plugin.cpp launcher.cpp launcher.h)
target_link_libraries(monetlauncherplugin PRIVATE monetlaunchcore Qt6::Qml)
target_compile_options(monetlauncherplugin PRIVATE -Wall -Wextra -Werror)
set_target_properties(monetlauncherplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/qml/org/monet/launcher")
configure_file(qmldir "${CMAKE_CURRENT_BINARY_DIR}/qml/org/monet/launcher/qmldir" COPYONLY)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
