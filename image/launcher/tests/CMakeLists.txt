find_package(Qt6 REQUIRED COMPONENTS Test Network)
find_package(KF6JobWidgets REQUIRED)
add_executable(launcher-test launcher-test.cpp)
target_include_directories(launcher-test PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(launcher-test PRIVATE monetlaunchcore Qt6::Test Qt6::Qml)
target_compile_options(launcher-test PRIVATE -Wall -Wextra -Werror)
add_dependencies(launcher-test monetlauncherplugin)
add_test(NAME desktop-launcher COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/run.sh $<TARGET_FILE:launcher-test>)
set_tests_properties(desktop-launcher PROPERTIES TIMEOUT 100 ENVIRONMENT
    "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QML_IMPORT_PATH=${CMAKE_BINARY_DIR}/qml")
add_executable(folder-test folder-test.cpp)
target_include_directories(folder-test PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(folder-test PRIVATE monetlaunchcore Qt6::Test Qt6::Network)
target_compile_options(folder-test PRIVATE -Wall -Wextra -Werror)
add_test(NAME transfer-folders COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/run.sh $<TARGET_FILE:folder-test>)
set_tests_properties(transfer-folders PROPERTIES TIMEOUT 100 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
add_executable(tracked-file-job tracked-file-job.cpp)
target_link_libraries(tracked-file-job PRIVATE Qt6::Gui KF6::KIOCore KF6::JobWidgets)
target_compile_options(tracked-file-job PRIVATE -Wall -Wextra -Werror)
