cmake_minimum_required(VERSION 3.24)
project(MonetRecordingClicks LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_SKIP_RPATH ON)
find_package(ECM REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(ECMQtDeclareLoggingCategory)
find_package(KWin CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Network Gui GuiPrivate DBus Widgets Qml Quick WaylandCompositor)
find_package(KF6I18n REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3)
set(dir "${UPSTREAM_SOURCE}/src/plugins/screencast")
file(GLOB sources CONFIGURE_DEPENDS "${dir}/*.cpp")
add_library(screencast MODULE ${sources})
set_target_properties(screencast PROPERTIES PREFIX "")
target_include_directories(screencast PRIVATE "${dir}" "${UPSTREAM_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}")
ecm_qt_declare_logging_category(screencast HEADER kwinscreencast_logging.h IDENTIFIER KWIN_SCREENCAST
    CATEGORY_NAME kwin_screencast DEFAULT_SEVERITY Warning)
target_link_options(screencast PRIVATE -Wl,-z,defs)
target_link_libraries(screencast PRIVATE KWin::kwin Qt6::Network Qt6::GuiPrivate Qt6::WaylandCompositor
    KF6::CoreAddons KF6::I18n PkgConfig::DRM PkgConfig::PipeWire)
