cmake_minimum_required(VERSION 3.22)
project(MonetWindowIdentity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui GuiPrivate DBus Widgets Qml)
add_library(monet_window_identity MODULE identity.cpp controls.cpp controls.h)
target_link_libraries(monet_window_identity PRIVATE KWin::kwin Qt6::DBus Qt6::Qml Qt6::GuiPrivate)
set_target_properties(monet_window_identity PROPERTIES PREFIX "")

find_package(KF6GlobalAccel CONFIG REQUIRED)
find_package(KF6I18n CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(XKBCOMMON REQUIRED IMPORTED_TARGET xkbcommon)
add_library(monet_plain_paste MODULE plainpaste.cpp)
target_link_libraries(monet_plain_paste PRIVATE KWin::kwin Qt6::GuiPrivate KF6::GlobalAccel KF6::I18n PkgConfig::XKBCOMMON)
set_target_properties(monet_plain_paste PROPERTIES PREFIX "")
find_package(KF6Service CONFIG REQUIRED)
add_library(monet_quit_guard MODULE quitguard.cpp)
target_link_libraries(monet_quit_guard PRIVATE KWin::kwin Qt6::GuiPrivate Qt6::DBus KF6::Service KF6::I18n PkgConfig::XKBCOMMON)
set_target_properties(monet_quit_guard PROPERTIES PREFIX "")

pkg_check_modules(WaylandServer REQUIRED IMPORTED_TARGET wayland-server)
add_library(monet_wheel_scroll MODULE wheelscroll.cpp)
target_link_libraries(monet_wheel_scroll PRIVATE KWin::kwin Qt6::GuiPrivate Qt6::DBus PkgConfig::XKBCOMMON PkgConfig::WaylandServer)
set_target_properties(monet_wheel_scroll PROPERTIES PREFIX "")
